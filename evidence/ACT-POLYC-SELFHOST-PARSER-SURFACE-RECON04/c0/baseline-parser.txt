# The suite harness (src/tests/run.HC:13-14) compiles each test with
# "../../hcc <file>" and WITHOUT --install-dir.  Without a writable
# /usr/local on this host, every test therefore fails with
#   "Failed to open file: /usr/local/include/tos.HH"

# HISTORICAL installation-prefix limitation recorded by

# carried forward.
#
# The remedy used here is the SAME hermetic pattern as
# scripts/quality/gate-push.sh: build the compiler with
# -DCMAKE_INSTALL_PREFIX=<prefix> so that the compile-time
# INSTALL_PREFIX matches the runtime prefix.
#

COMMAND=INSTALL_PREFIX=/tmp/recon04-baseline/prefix make all
COMMAND=INSTALL_PREFIX=/tmp/recon04-baseline/prefix TEST_PREFIX=/tmp/recon04-baseline/prefix make test-prefix-install

#   COMPILER_PREFIX=/tmp/recon04-baseline/prefix
