COMMAND=make -n production-hcc | grep -c 'parser-default-arm-subject.o'
   1

# ADVERSE: the subject object is REMOVED from the prerequisite list

#         list; make -n production-hcc | grep -c 'parser-default-arm-subject.o')
   0

ADVERSE_DETECTOR=MISSING_SUBJECT_PREREQUISITE

# The restoration was VERIFIED: the live Makefile's subject line is

COMMAND=grep -c 'SLICE08_DEFAULT_ARM_OBJ) \' Makefile
   6
RESTORATION_VERIFIED=YES

# The same probe is applied to the NEW subject at C2, and its result
# is recorded at c2/subject-build-graph-proof.txt.
