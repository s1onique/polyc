# COMMAND=./build/recon05/parser-surface-recon05-verify IDENTITY c1/candidate-identity.tsv src/parser.c src/prslib.c src/prsutil.c src/ast.c
#
#   IDENTITY_ROWS=24
#   DUPLICATE_ID=0
#   INVALID_RANGE=0
#   SOURCE_HASH_MISMATCH=0
#   MISSING_FUNCTION=0
#   OVERLAPPING_IDENTITY=0
#   MALFORMED_FACT=0
#   STATUS=PASS
#
# The checker recomputes the SHA-256 of every candidate range from
# the live source bytes and compares it to the frozen digest.  It
# also resolves every candidate to a live function and rejects any
# two candidates whose ranges conflict.
#
# THE DEPENDENCE CHECK
# COMMAND=./build/recon05/parser-surface-recon05-verify DEPENDS c1/candidate-identity.tsv src/parser.c src/prslib.c src/prsutil.c src/ast.c
#
#   DEPENDS_CHECKED=24
#   MUTABLE_HOST_STATE_DEPENDENCE=1
#   NO_MUTABLE_HOST_STATE_DEPENDENCE=23
#   UNBOUND_DEPENDENCY=0
#
# The single MUTABLE_HOST_STATE_DEPENDENCE is SUB-008 (the whole
# statement parser), whose body writes cc->current_recovery.  That
# is the CORRECT result and it is exactly why SUB-008 is
# ineligible.  Every other candidate decision is provably free of
# the mutable symbol table, the recovery point, the host AST graph
# and the token stream.
