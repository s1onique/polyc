#    every row)

# --- STEP 4: the fixture counts, DERIVED from the corrected matrix -------
# The accept/reject totals and the group breakdown are computed from the
# ROWS of the corrected matrix, not entered by hand.
COMMAND_1C=awk -F'\t' '
function grp(id) {
  if (id == "F001") return "NULL_EOF"
  if (id == "F002" || id == "F003" || id == "F004") return "IDENT"
  if (id == "F005" || id == "F006" || id == "F041") return "STR"
  if (id >= "F007" && id <= "F017") return "PUNCT"
  if (id >= "F018" && id <= "F035") return "KEYWORD"
  return "UNKNOWN_KIND"
}
NR>1 && $1 ~ /^F[0-9]/ {
  n++
  g = grp($1)
  tot[g]++
  if ($6 == 1) { acc++; accg[g]++ } else { rej++; rejg[g]++ }
}
END {
  print "rows=" n
  print "accept=" acc " reject=" rej
  for (g in tot) print "group " g " total=" tot[g] " accept=" accg[g]+0 " reject=" rejg[g]+0
}' evidence/ACT-POLYC-SELFHOST-PARSER-SLICE-RECON02-CORRECTION01/c1/c1-corrected-fixture-model.tsv
