#!/bin/sh
# Demonstrate that the seed silently skips the two fixtures.
# We invoke the existing seed pairing logic directly.
REPO="$1"
ACTS_DIR="$REPO/docs/acts"
EVID_DIR="$REPO/evidence"
fail=0
for act_path in "$ACTS_DIR"/ACT-*.md "$REPO/evidence/factory-status-reconciliation/red-3a1/fixture/managed/extra-scope-factory-NON_INFERRED/docs/acts/ACT-POLYC-FACTORY-NONINFERRED.md" "$REPO/evidence/factory-status-reconciliation/red-3b/fixture/managed/missing-handoff-ACT/docs/acts/ACT-POLYC-FACTORY-MISSINGHO.md"; do
    [ -f "$act_path" ] || continue
    fname="$(basename "$act_path")"
    # Use the seed's act_to_evid_dir function
    base="${fname%.md}"
    base="${base#ACT-POLYC-}"
    case "$base" in
        LLVM-*)    tail="${base#LLVM-}";;
        IR-*)      tail="${base#IR-}";;
        FACTORY-*) tail="${base#FACTORY-}";;
        *)         tail="$base";;
    esac
    tail="$(echo "$tail" | tr '[:upper:]' '[:lower:]')"
    if [ "${tail#core}" != "$tail" ]; then
        dir="llvmspike01-$tail"
    elif [ "${tail#boundary}" != "$tail" ]; then
        dir="ir-$tail"
    elif [ "${tail#factory-}" != "$tail" ]; then
        dir="$tail"
    else
        dir="$tail"
    fi
    handoff="$EVID_DIR/$dir/HANDOFF.md"
    if [ -f "$handoff" ]; then
        echo "DISCOVER  $fname  ->  $handoff"
    else
        echo "OMIT      $fname  ->  $handoff  (NOT FOUND, seed skips)"
    fi
done
