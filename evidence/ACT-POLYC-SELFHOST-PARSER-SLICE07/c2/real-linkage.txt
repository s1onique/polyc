# The `U` is the UNDEFINED-reference marker: the production parser

# PRODUCTION_PARSER_OBJECT_IMPORTS_THE_SYMBOL=YES

# ---------------------------------------------------------------------------
# PROOF 2: THE SUBJECT OBJECT DEFINES THE SYMBOL

# The `T` is the global TEXT marker: the subject EXPORTS the symbol.
#
# SUBJECT_OBJECT_DEFINES_THE_SYMBOL=YES
# SUBJECT_EXPORTED_GLOBAL_TEXT_SYMBOL_COUNT=1
#
# The subject's ONLY exported global text symbol is the component's.
# There is no unexpected extra exported text.
#
