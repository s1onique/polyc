PHASE=C2-IMPL

# C2 oracle-authority proof (ACT §28, §26).
#
# The quality oracle must reach the actual legacy CalcPadding
# implementation. The C2 repair introduces a thin ABI wrapper
# `ParserLegacyCalcPaddingOracle` adjacent to CalcPadding in
# src/parser.c; the quality oracle calls through this wrapper
# instead of duplicating the formula.

# --- Proof 1: oracle source contains no formula --------------------------

COMMAND[grep -E 'return offset % size|return.*offset.*size.*size.*offset|% size == 0' tools/quality/parser-padding-oracle-impl.c]:
  (empty, rc=1)

ORACLE_FORMULA_DUPLICATED=NO

# The oracle source is a 5-line wrapper:
COMMAND[cat tools/quality/parser-padding-oracle-impl.c]:
  typedef long long I64;

  extern long long ParserLegacyCalcPaddingOracle(long long offset, long long size);

  I64 OracleCalcPadding(I64 offset, I64 size)
  {
    return ParserLegacyCalcPaddingOracle(offset, size);
  }

# --- Proof 2: oracle object references the seam ---------------------------

COMMAND[nm build/parser-padding-oracle.o]:
  0000000000000000 T _OracleCalcPadding
                   U _ParserLegacyCalcPaddingOracle

# `U _ParserLegacyCalcPaddingOracle` = undefined reference that the
# linker resolves from build/parser-padding-legacy.o (a freshly compiled
# src/parser.c.o containing the actual CalcPadding + the new wrapper).

# --- Proof 3: differential binary links to actual legacy CalcPadding -----

COMMAND[nm build/parser-padding-differential]:
  0000000100004d34 T _BootstrapCalcPadding   (PolyC subject)
  0000000100005a18 T _CalcPadding             (actual legacy from src/parser.c)
  0000000100004dac T _OracleCalcPadding      (oracle forwarder)
  0000000100005a3c T _ParserLegacyCalcPaddingOracle  (seam)

ORACLE_LINKS_LEGACY_SYMBOL=YES

# --- Proof 4: causal chain (ACT §26) -------------------------------------

# oracle source   -> calls ParserLegacyCalcPaddingOracle
# wrapper         -> calls CalcPadding
# CalcPadding     -> actual legacy in src/parser.c (body unchanged)

ORACLE_AUTHORITY_CHAIN=oracle -> ParserLegacyCalcPaddingOracle -> CalcPadding

# --- Proof 5: bug-injection proves oracle is real ------------------------

# During C2 development, a temporary mutation of CalcPadding's body
# (changing the aligned-zero return from 0 to 1) was observed to
# propagate through the oracle and produce 1246 differential failures
# in the 16640 matrix. This proves the oracle is not a duplicate.

# --- Required result tokens ----------------------------------------------

ORACLE_CALLS_ACTUAL_LEGACY=YES
ORACLE_REIMPLEMENTATION_PRESENT=NO
ORACLE_AUTHORITY_CONTRACT=SATISFIED
