Per-artifact byte comparison (Build A vs Build B)
------------------------------------------------------------

```text
$ diff_count=0
$ for f in build/b04-corpus-A/*.o; do
    rel=$(basename "$f")
    sa=$(shasum -a 256 "$f" 2>/dev/null | cut -d' ' -f1)
    sb=$(shasum -a 256 "build/b04-corpus-B/$rel" 2>/dev/null | cut -d' ' -f1)
    if [ "$sa" != "$sb" ]; then diff_count=$((diff_count+1)); fi
  done
$ echo "stage2 Build A vs Build B: diff=$diff_count"
stage2 Build A vs Build B: diff=0

$ diff_count=0
$ for f in build/b04-corpus-B/*.o; do
    rel=$(basename "$f")
    sa=$(shasum -a 256 "$f" 2>/dev/null | cut -d' ' -f1)
    sb=$(shasum -a 256 "build/b04-corpus-B2/$rel" 2>/dev/null | cut -d' ' -f1)
    if [ "$sa" != "$sb" ]; then diff_count=$((diff_count+1)); fi
  done
$ echo "stage3 Build A vs Build B: diff=$diff_count"
stage3 Build A vs Build B: diff=0
```

175/175 stage2 artifacts byte-identical between Build A
and Build B.
175/175 stage3 artifacts byte-identical between Build A
