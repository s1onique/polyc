#             (e.g. ./build/hcc-bootstrap02-build/CMakeFiles/hcc-bootstrap02.dir)
#   argv[6] = stage1 objdir fallback
#             (e.g. ./build/hcc-bootstrap02-build/CMakeFiles/hcc-bootstrap02.dir)
