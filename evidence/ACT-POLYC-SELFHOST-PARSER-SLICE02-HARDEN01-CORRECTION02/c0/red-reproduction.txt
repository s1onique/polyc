# RED C -- THE DRIVER'S ROW-SEPARATION RISK
# ===================================================================
#
# EditAppendLine() in tools/quality/parser-control-flow-token-
# negative-controls.HC appends at g_src_len and does not insert a
# separating newline:
#
#     U0 EditAppendLine(U8 *text)
#     {
#       g_cut_a = g_src_len;
#       g_cut_b = g_src_len;
#       ...
#       g_repl[tl] = '\n';
#
# On a source that does not end with a newline, the appended row JOINS
# the last existing row.
